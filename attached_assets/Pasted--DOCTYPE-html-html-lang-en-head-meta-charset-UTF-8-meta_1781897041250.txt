// ── Calendar ──
const MONTHS = ['January','February','March','April','May','June','July','August','September','October','November','December'];

function renderCalendar() {
  document.getElementById('cal-month-label').textContent = `${MONTHS[currentMonth]} ${currentYear}`;

  const firstDay    = new Date(currentYear, currentMonth, 1).getDay();
  const daysInMonth = new Date(currentYear, currentMonth + 1, 0).getDate();
  const daysInPrev  = new Date(currentYear, currentMonth, 0).getDate();
  const taskDates   = Object.keys(tasks);

  const grid = document.getElementById('cal-days');
  grid.innerHTML = '';

  // Prev month tail
  for (let i = firstDay - 1; i >= 0; i--) {
    const d = el('div', 'cal-day other-month', daysInPrev - i);
    grid.appendChild(d);
  }

  // Current month days
  for (let day = 1; day <= daysInMonth; day++) {
    const ds = `${currentYear}-${pad(currentMonth+1)}-${pad(day)}`;
    const d  = el('div', 'cal-day', day);
    if (ds === TODAY)        d.classList.add('today');
    if (ds === selectedDate) d.classList.add('selected');
    if (taskDates.includes(ds)) d.classList.add('has-tasks');
    d.onclick = () => selectDate(ds);
    grid.appendChild(d);
  }

  // Next month fill
  const total = Math.ceil((firstDay + daysInMonth) / 7) * 7;
  for (let i = 1; i <= total - firstDay - daysInMonth; i++) {
    grid.appendChild(el('div', 'cal-day other-month', i));
  }
}

function changeMonth(dir) {
  currentMonth += dir;
  if (currentMonth < 0)  { currentMonth = 11; currentYear--; }
  if (currentMonth > 11) { currentMonth = 0;  currentYear++; }
  renderCalendar();
}

function selectDate(ds) {

        </div>
        <div class="upcoming-badge tag-${ts[0]?.cat||'chore'}">${done}/${ts.length}</div>

  }).join('');
}

function toggleTask(id) {
  const list = tasks[selectedDate];
  if (!list) return;
  const t = list.find(x => x.id === id);
  if (!t) return;
  t.done = !t.done;
  renderTasks(); renderCalendar(); renderUpcoming();
  if (t.done) showNotif(`✓ "${t.title}" done!${t.points ? ' You earned ' + t.points + '!' : ''}`, 'success');
}

function removeTask(id) {
  if (!tasks[selectedDate]) return;
  tasks[selectedDate] = tasks[selectedDate].filter(t => t.id !== id);
  if (!tasks[selectedDate].length) delete tasks[selectedDate];
  renderTasks(); renderCalendar(); renderUpcoming();
  showNotif('Task removed', 'warning');
}

function addTask() {
  const title = document.getElementById('f-title').value.trim();
  if (!title) { showNotif('Please enter a task name', 'warning'); return; }

  const raw  = document.getElementById('f-time').value;
  let timeStr = '';
  if (raw) {
    const [h, m] = raw.split(':');
    const hr = parseInt(h);
    timeStr = `${hr > 12 ? hr-12 : hr||12}:${m} ${hr >= 12 ? 'PM' : 'AM'}`;
  }

  const newTask = {
    id: Date.now(),
    title,
    cat:    document.getElementById('f-cat').value,
    time:   timeStr,
    note:   document.getElementById('f-note').value.trim(),
    points: document.getElementById('f-points').value,
    done:   false,
  };

  if (!tasks[selectedDate]) tasks[selectedDate] = [];
  tasks[selectedDate].push(newTask);

  closeModal();
  renderTasks(); renderCalendar(); renderUpcoming();
  showNotif('Task added for Emma ✓', 'success');
}

let notifTimer;

}

function el(tag, cls, text) {

  d.className = cls;
  if (text !== undefined) d.textContent = text;
  return d;
}
