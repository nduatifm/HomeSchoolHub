const CustomTooltip = ({ active, payload, label }: any) => {
    if (active && payload && payload.length) {
        return (
            <div className="bg-white/95 backdrop-blur-md p-3 rounded-xl shadow-[0_8px_30px_rgb(0,0,0,0.12)] border border-white/80 text-xs min-w-[120px] z-50">
                {label && <p className="font-bold text-gray-800 mb-2 border-b border-gray-100 pb-1.5">{label}</p>}
                {!label && payload[0].payload.name && <p className="font-bold text-gray-800 mb-2 border-b border-gray-100 pb-1.5">{payload[0].payload.name}</p>}
                {payload.map((entry: any, index: number) => {
                    const name = entry.name === 'val1' ? 'Current' : entry.name === 'val2' ? 'Previous' : entry.name === 'value' ? 'Amount' : entry.name;
                    return (
                        <div key={index} className="flex items-center justify-between gap-4 mt-1.5">
                            <div className="flex items-center gap-2">
                                <div className="w-2.5 h-2.5 rounded-full shadow-sm" style={{ backgroundColor: entry.color || entry.fill }}></div>
                                <span className="text-gray-600 font-medium">{name}</span>
                            </div>
                            <span className="font-bold text-gray-900">
                                ${typeof entry.value === 'number' ? entry.value.toLocaleString() : entry.value}
                            </span>
                        </div>
                    )
                })}
            </div>
        );
    }
    return null;
};
