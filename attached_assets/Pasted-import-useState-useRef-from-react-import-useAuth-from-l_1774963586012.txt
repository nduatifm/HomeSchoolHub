  Form,

  FormMessage,
} from "@/components/ui/form";

  SelectValue,
} from "@/components/ui/select";
import {
  AlertDialog,
  AlertDialogAction,
  AlertDialogCancel,
  AlertDialogContent,
  AlertDialogDescription,
  AlertDialogFooter,
  AlertDialogHeader,
  AlertDialogTitle,
} from "@/components/ui/alert-dialog";
import { useToast } from "@/hooks/use-toast";
import { Loader2, User, Camera, AlertTriangle, Mail, Clock, X } from "lucide-react";
import { apiRequest } from "@/lib/queryClient";
import { SEO } from "@/components/seo";
import { useUpload } from "@/hooks/use-upload";
import { AvatarCropper } from "@/components/avatar-cropper";

function FlagImage({
  countryCode,
  className = "",
}: {
  countryCode: string;
  className?: string;
}) {
  return (
    <img
      src={`https://flagcdn.com/w40/${countryCode}.png`}
      srcSet={`https://flagcdn.com/w80/${countryCode}.png 2x`}
      alt=""
      className={`inline-block rounded-sm ${className}`}
      style={{ width: "24px", height: "18px", objectFit: "cover" }}
    />
  );
}

const profileSchema = z.object({
  name: z.string().min(1, "Name is required").max(100, "Name is too long"),
  email: z.string().email("Invalid email address"),
  photoUrl: z.string().optional().or(z.literal("")),
  currency: z.string().default("USD"),
});

type ProfileFormValues = z.infer<typeof profileSchema>;

export default function SettingsPage() {
  const { user, refreshUser } = useAuth();
  const { toast } = useToast();
  const queryClient = useQueryClient();
  const fileInputRef = useRef<HTMLInputElement>(null);
  const [previewUrl, setPreviewUrl] = useState<string | null>(null);
  const [editingField, setEditingField] = useState<string | null>(null);
  const [cropImageSrc, setCropImageSrc] = useState<string | null>(null);
  const [showCropper, setShowCropper] = useState(false);
  const [showEmailChangeDialog, setShowEmailChangeDialog] = useState(false);
  const [newEmailInput, setNewEmailInput] = useState("");

  const { data: pendingEmailData, refetch: refetchPendingEmail } = useQuery<{
    hasPendingChange: boolean;
    pendingEmail?: string;
    expiresAt?: string;
  }>({
    queryKey: ["/api/users/pending-email"],
    enabled: !!user,
  });

  const requestEmailChangeMutation = useMutation({
    mutationFn: async (newEmail: string) => {
      const response = await apiRequest("POST", "/api/users/request-email-change", { newEmail });
      return response.json();
    },
    onSuccess: (data) => {
      toast({
        title: "Verification email sent",
        description: data.message || "Please check your new email address for the verification link.",
      });
      setShowEmailChangeDialog(false);
      setEditingField(null);
      setNewEmailInput("");
      refetchPendingEmail();
    },
    onError: (error: any) => {
      toast({
        title: "Failed to request email change",
        description: error.message || "Something went wrong. Please try again.",
        variant: "destructive",
      });
    },
  });

  const cancelEmailChangeMutation = useMutation({
    mutationFn: async () => {
      const response = await apiRequest("POST", "/api/users/cancel-email-change", {});
      return response.json();
    },
    onSuccess: () => {
      toast({
        title: "Email change cancelled",
        description: "Your pending email change has been cancelled.",
      });
      refetchPendingEmail();
    },
    onError: (error: any) => {
      toast({
        title: "Failed to cancel",
        description: error.message || "Something went wrong. Please try again.",
        variant: "destructive",
      });
    },
  });

  const form = useForm<ProfileFormValues>({
    resolver: zodResolver(profileSchema),
    defaultValues: {
      name: user?.name || "",
      email: user?.email || "",
      photoUrl: user?.photoUrl || "",
      currency: (user as any)?.currency || "USD",
    },
  });

  const { uploadFile, isUploading } = useUpload({
    onSuccess: (response) => {
      const photoUrl = response.secureUrl;
      form.setValue("photoUrl", photoUrl);
      toast({
        title: "Photo uploaded",
        description: "Your profile picture has been updated.",
      });
      saveField("photoUrl", photoUrl);
    },
    onError: (error) => {
      toast({
        title: "Upload failed",
        description:
          error.message || "Failed to upload photo. Please try again.",
        variant: "destructive",
      });
      setPreviewUrl(null);
    },
  });

  const updateProfileMutation = useMutation({
    mutationFn: async (data: Partial<ProfileFormValues>) => {
      const response = await apiRequest("PATCH", "/api/users/profile", data);
      return response.json();
    },
    onSuccess: async () => {
      toast({
        title: "Saved",
        description: "Your changes have been saved.",
      });
      if (refreshUser) {
        await refreshUser();
      }
      queryClient.invalidateQueries({ queryKey: ["/api/users/me"] });
      setPreviewUrl(null);
      setEditingField(null);
    },
    onError: (error: any) => {
      toast({
        title: "Update failed",
        description: error.message || "Failed to update. Please try again.",
        variant: "destructive",
      });
    },
  });

  const saveField = (fieldName: string, value: string) => {
    const updates: Partial<ProfileFormValues> = { [fieldName]: value };
    updateProfileMutation.mutate(updates);
  };

  const handleSaveField = (fieldName: keyof ProfileFormValues) => {
    const value = form.getValues(fieldName);
    if (fieldName === "name" && value === user?.name) {
      setEditingField(null);
      return;
    }
    if (fieldName === "email" && value === user?.email) {
      setEditingField(null);
      return;
    }
    if (
      fieldName === "currency" &&
      value === ((user as any)?.currency || "USD")
    ) {
      setEditingField(null);
      return;
    }
    saveField(fieldName, value as string);
  };

  const handleFileChange = (e: React.ChangeEvent<HTMLInputElement>) => {
    const file = e.target.files?.[0];
    if (!file) return;

    e.target.value = "";

    if (!file.type.startsWith("image/")) {
      toast({
        title: "Invalid file type",
        description: "Please select an image file.",
        variant: "destructive",
      });
      return;
    }

    if (file.size > 10 * 1024 * 1024) {
      toast({
        title: "File too large",
        description: "Please select an image smaller than 10MB.",
        variant: "destructive",
      });
      return;
    }

    const reader = new FileReader();
    reader.onloadend = () => {
      setCropImageSrc(reader.result as string);
      setShowCropper(true);
    };
    reader.readAsDataURL(file);
  };

  const handleCropComplete = async (croppedBlob: Blob) => {
    const croppedFile = new File([croppedBlob], "avatar.jpg", { type: "image/jpeg" });
    const result = await uploadFile(croppedFile);
    if (result) {
      setShowCropper(false);
      setCropImageSrc(null);
    }
  };

  const getInitial = () => {
    if (user?.name) {
      return user.name.charAt(0).toUpperCase();
    }
    if (user?.email) {
      return user.email.charAt(0).toUpperCase();
    }
    return <User className="h-6 w-6" />;
  };

  const displayPhotoUrl =
    previewUrl || form.watch("photoUrl") || user?.photoUrl;
  const selectedCurrency = CURRENCIES.find(
    (c) => c.code === form.watch("currency"),
  );

  if (!user) {
    return (
      <div className="min-h-screen bg-gray-50">
        <div className="max-w-2xl mx-auto py-12 px-4 sm:px-6">
          <div className="h-8 w-32 bg-slate-200 rounded animate-pulse mb-8" />
          <div className="bg-white border border-gray-200 rounded-xl divide-y divide-gray-100 shadow-sm">
            <div className="px-6 py-6">
              <div className="flex items-center gap-5">
                <div className="w-20 h-20 rounded-full bg-slate-200 animate-pulse" />
                <div className="space-y-2">
                  <div className="h-5 w-32 bg-slate-200 rounded animate-pulse" />
                  <div className="h-4 w-24 bg-slate-200 rounded animate-pulse" />
                </div>
              </div>
            </div>
            <div className="px-6 py-6 space-y-4">
              <div className="h-4 w-20 bg-slate-200 rounded animate-pulse" />
              <div className="h-10 w-full bg-slate-200 rounded animate-pulse" />
            </div>
            <div className="px-6 py-6 space-y-4">
              <div className="h-4 w-24 bg-slate-200 rounded animate-pulse" />
              <div className="h-10 w-full bg-slate-200 rounded animate-pulse" />
            </div>
          </div>
        </div>
      </div>
    );
  }

  return (
    <>
      <SEO
        title="Settings | MoneyTallyHQ"
        description="Manage your MoneyTallyHQ account settings, update your profile information, and customize your experience."
      />
      <div className="min-h-screen bg-gray-50">
        <div className="max-w-2xl mx-auto py-12 px-4 sm:px-6">
          <h1 className="text-2xl font-semibold text-gray-900 mb-8">Account</h1>

          <Form {...form}>
            <div className="bg-white border border-gray-200 rounded-xl divide-y divide-gray-100 shadow-sm">
              <div className="px-6 py-6">
                <div className="flex items-center gap-5">
                  <input
                    ref={fileInputRef}
                    type="file"
                    accept="image/*"
                    onChange={handleFileChange}
                    className="hidden"
                    data-testid="input-photo-file"
                  />
                  <button
                    type="button"
                    onClick={() => fileInputRef.current?.click()}
                    disabled={isUploading}
                    className="relative group cursor-pointer"
                    data-testid="button-upload-photo"
                  >
                    <Avatar className="h-20 w-20 border-2 border-gray-100 group-hover:border-gray-200 transition-all">
                      <AvatarImage
                        src={displayPhotoUrl || undefined}
                        alt={user.name || "User"}
                      />
                      <AvatarFallback className="bg-gradient-to-br from-blue-50 to-indigo-50 text-gray-700 text-lg font-medium">
                        {getInitial()}
                      </AvatarFallback>
                    </Avatar>
                    {isUploading ? (
                      <div className="absolute inset-0 flex items-center justify-center bg-black/40 rounded-full backdrop-blur-sm">
                        <Loader2 className="h-6 w-6 animate-spin text-white" />
                      </div>
                    ) : (
                      <div className="absolute inset-0 flex items-center justify-center bg-black/0 group-hover:bg-black/40 rounded-full transition-all backdrop-blur-0 group-hover:backdrop-blur-sm">
                        <Camera className="h-5 w-5 text-white opacity-0 group-hover:opacity-100 transition-opacity" />
                      </div>
                    )}
                  </button>
                  <div>
                    <p className="text-xs font-medium text-gray-500 uppercase tracking-wider mb-1">
                      Profile
                    </p>
                    <p className="text-lg font-medium text-gray-900">
                      {user.name}
                    </p>
                  </div>
                </div>
              </div>

              <div className="px-6 py-5">
                <div className="flex items-start justify-between gap-4">
                  <div className="flex-1 min-w-0">
                    <p className="text-xs font-medium text-gray-500 uppercase tracking-wider mb-2">
                      Name
                    </p>
                    {editingField === "name" ? (
                      <FormField
                        control={form.control}
                        name="name"
                        render={({ field }) => (
                          <FormItem>
                            <FormControl>
                              <div className="flex gap-2">
                                <Input
                                  {...field}
                                  className="h-10 border-gray-300 focus:border-blue-500 focus:ring-blue-500"
                                  data-testid="input-name"
                                  autoFocus
                                />
                                <Button
                                  type="button"
                                  size="sm"
                                  onClick={() => handleSaveField("name")}
                                  disabled={updateProfileMutation.isPending}
                                  className="px-5 h-10 bg-gray-900 hover:bg-gray-700 text-white"
                                >
                                  {updateProfileMutation.isPending ? (
                                    <Loader2 className="h-4 w-4 animate-spin" />
                                  ) : (
                                    "Save"
                                  )}
                                </Button>
                                <Button
                                  type="button"
                                  variant="ghost"
                                  size="sm"
                                  onClick={() => {
                                    form.setValue("name", user.name || "");
                                    setEditingField(null);
                                  }}
                                  className="h-10 text-gray-700 hover:bg-gray-100 hover:text-gray-900"
                                >
                                  Cancel
                                </Button>
                              </div>
                            </FormControl>
                            <FormMessage />
                          </FormItem>
                        )}
                      />
                    ) : (
                      <p className="text-base text-gray-900 truncate">
                        {user.name}
                      </p>
                    )}
                  </div>
                  {editingField !== "name" && (
                    <button
                      type="button"
                      className="rounded-full px-5 h-9 text-sm font-medium text-gray-700 border border-gray-300 bg-white hover:bg-gray-100 hover:border-gray-400 transition-colors flex-shrink-0"
                      onClick={() => setEditingField("name")}
                    >
                      Edit
                    </button>
                  )}
                </div>
              </div>

              <div className="px-6 py-5">
                <div className="flex items-start justify-between gap-4">
                  <div className="flex-1 min-w-0">
                    <p className="text-xs font-medium text-gray-500 uppercase tracking-wider mb-2">
                      Email
                    </p>
                    {editingField === "email" ? (
                      <div className="space-y-3">
                        <div className="flex gap-2">
                          <Input
                            type="email"
                            value={newEmailInput}
                            onChange={(e) => setNewEmailInput(e.target.value)}
                            placeholder="Enter new email address"
                            className="h-10 border-gray-300 focus:border-blue-500 focus:ring-blue-500"
                            data-testid="input-email"
                            autoFocus
                          />
                          <Button
                            type="button"
                            size="sm"
                            onClick={() => {
                              if (newEmailInput && newEmailInput !== user?.email) {
                                setShowEmailChangeDialog(true);
                              } else if (newEmailInput === user?.email) {
                                setEditingField(null);
                                setNewEmailInput("");
                              }
                            }}
                            disabled={!newEmailInput || requestEmailChangeMutation.isPending}
                            className="px-5 h-10 bg-gray-900 hover:bg-gray-700 text-white"
                          >
                            Continue
                          </Button>
                          <Button
                            type="button"
                            variant="ghost"
                            size="sm"
                            onClick={() => {
                              setEditingField(null);
                              setNewEmailInput("");
                            }}
                            className="h-10 text-gray-700 hover:bg-gray-100 hover:text-gray-900"
                          >
                            Cancel
                          </Button>
                        </div>
                      </div>
                    ) : (
                      <div className="space-y-2">
                        <p className="text-base text-gray-900 truncate">
                          {user.email}
                        </p>
                        {pendingEmailData?.hasPendingChange && (
                          <div className="flex items-center gap-2 p-3 bg-amber-50 border border-amber-200 rounded-lg">
                            <Clock className="h-4 w-4 text-amber-600 flex-shrink-0" />
                            <div className="flex-1 min-w-0">
                              <p className="text-sm text-amber-800">
                                Pending change to: <span className="font-medium">{pendingEmailData.pendingEmail}</span>
                              </p>
                              <p className="text-xs text-amber-600 mt-0.5">
                                Check your email for the verification link
                              </p>
                            </div>
                            <button
                              type="button"
                              onClick={() => cancelEmailChangeMutation.mutate()}
                              disabled={cancelEmailChangeMutation.isPending}
                              className="p-1 text-amber-600 hover:text-amber-800 hover:bg-amber-100 rounded transition-colors"
                              title="Cancel pending change"
                              data-testid="button-cancel-email-change"
                            >
                              {cancelEmailChangeMutation.isPending ? (
                                <Loader2 className="h-4 w-4 animate-spin" />
                              ) : (
                                <X className="h-4 w-4" />
                              )}
                            </button>
                          </div>
                        )}
                      </div>
                    )}
                  </div>
                  {editingField !== "email" && !pendingEmailData?.hasPendingChange && (
                    <button
                      type="button"
                      className="rounded-full px-5 h-9 text-sm font-medium text-gray-700 border border-gray-300 bg-white hover:bg-gray-100 hover:border-gray-400 transition-colors flex-shrink-0"
                      onClick={() => setEditingField("email")}
                    >
                      Edit
                    </button>
                  )}
                </div>
              </div>

              <div className="px-6 py-5">
                <div className="flex items-start justify-between gap-4">
                  <div className="flex-1 min-w-0">
                    <p className="text-xs font-medium text-gray-500 uppercase tracking-wider mb-2">
                      Currency
                    </p>
                    {editingField === "currency" ? (
                      <FormField
                        control={form.control}
                        name="currency"
                        render={({ field }) => (
                          <FormItem>
                            <div className="flex gap-2 items-start">
                              <Select
                                onValueChange={field.onChange}
                                value={field.value}
                              >
                                <FormControl>
                                  <SelectTrigger
                                    className="w-[280px] h-10 border-gray-300 focus:border-blue-500 focus:ring-blue-500"
                                    data-testid="select-currency"
                                  >
                                    {selectedCurrency ? (
                                      <div className="flex items-center gap-2">
                                        <FlagImage
                                          countryCode={
                                            selectedCurrency.countryCode
                                          }
                                        />
                                        <span>
                                          {selectedCurrency.code} -{" "}
                                          {selectedCurrency.name}
                                        </span>
                                      </div>
                                    ) : (
                                      <SelectValue placeholder="Select currency" />
                                    )}
                                  </SelectTrigger>
                                </FormControl>
                                <SelectContent className="max-h-[280px]">
                                  {CURRENCIES.map((currency) => (
                                    <SelectItem
                                      key={currency.code}
                                      value={currency.code}
                                      data-testid={`currency-option-${currency.code}`}
                                    >
                                      <div className="flex items-center gap-2">
                                        <FlagImage
                                          countryCode={currency.countryCode}
                                        />
                                        <span>
                                          {currency.code} - {currency.name}
                                        </span>
                                      </div>
                                    </SelectItem>
                                  ))}
                                </SelectContent>
                              </Select>
                              <Button
                                type="button"
                                size="sm"
                                onClick={() => handleSaveField("currency")}
                                disabled={updateProfileMutation.isPending}
                                className="px-5 h-10 bg-gray-900 hover:bg-gray-700 text-white"
                              >
                                {updateProfileMutation.isPending ? (
                                  <Loader2 className="h-4 w-4 animate-spin" />
                                ) : (
                                  "Save"
                                )}
                              </Button>
                              <Button
                                type="button"
                                variant="ghost"
                                size="sm"
                                onClick={() => {
                                  form.setValue(
                                    "currency",
                                    (user as any)?.currency || "USD",
                                  );
                                  setEditingField(null);
                                }}
                                className="h-10 text-gray-700 hover:bg-gray-100 hover:text-gray-900"
                              >
                                Cancel
                              </Button>
                            </div>
                            <FormMessage />
                          </FormItem>
                        )}
                      />
                    ) : (
                      <div className="flex items-center gap-2">
                        {selectedCurrency && (
                          <FlagImage
                            countryCode={selectedCurrency.countryCode}
                          />
                        )}
                        <p className="text-base text-gray-900">
                          {selectedCurrency
                            ? `${selectedCurrency.code} - ${selectedCurrency.name}`
                            : "USD - US Dollar"}
                        </p>
                      </div>
                    )}
                  </div>
                  {editingField !== "currency" && (
                    <button
                      type="button"
                      className="rounded-full px-5 h-9 text-sm font-medium text-gray-700 border border-gray-300 bg-white hover:bg-gray-100 hover:border-gray-400 transition-colors flex-shrink-0"
                      onClick={() => setEditingField("currency")}
                    >
                      Edit
                    </button>
                  )}
                </div>
              </div>
            </div>
          </Form>
        </div>
      </div>

      <AlertDialog open={showEmailChangeDialog} onOpenChange={setShowEmailChangeDialog}>
        <AlertDialogContent className="max-w-md">
          <AlertDialogHeader>
            <div className="flex items-center gap-3 mb-2">
              <div className="p-2 bg-amber-100 rounded-full">
                <AlertTriangle className="h-5 w-5 text-amber-600" />
              </div>
              <AlertDialogTitle>Change Email Address</AlertDialogTitle>
            </div>
            <AlertDialogDescription className="space-y-3 text-left">
              <p>
                You are about to change your email address from{" "}
                <span className="font-medium text-gray-900">{user?.email}</span> to{" "}
                <span className="font-medium text-gray-900">{newEmailInput}</span>.
              </p>
              <div className="bg-gray-50 border border-gray-200 rounded-lg p-3 space-y-2">
                <p className="text-sm font-medium text-gray-900">Please note:</p>
                <ul className="text-sm text-gray-600 space-y-1 list-disc list-inside">
                  <li>A verification link will be sent to your new email</li>
                  <li>Your current email will remain active until you verify</li>
                  <li>After verification, you'll sign in with your new email</li>
                  <li>The verification link expires in 24 hours</li>
                </ul>
              </div>
            </AlertDialogDescription>
          </AlertDialogHeader>
          <AlertDialogFooter>
            <AlertDialogCancel 
              onClick={() => setShowEmailChangeDialog(false)}
              className="border-gray-300"
            >
              Cancel
            </AlertDialogCancel>
            <AlertDialogAction
              onClick={() => requestEmailChangeMutation.mutate(newEmailInput)}
              disabled={requestEmailChangeMutation.isPending}
              className="bg-gray-900 hover:bg-gray-700 text-white"
              data-testid="button-confirm-email-change"
            >
              {requestEmailChangeMutation.isPending ? (
                <>
                  <Loader2 className="h-4 w-4 animate-spin mr-2" />
                  Sending...
                </>
              ) : (
                <>
                  <Mail className="h-4 w-4 mr-2" />
                  Send Verification Email
                </>
              )}
            </AlertDialogAction>
          </AlertDialogFooter>
        </AlertDialogContent>
      </AlertDialog>

      {cropImageSrc && (
        <AvatarCropper
          open={showCropper}
          imageSrc={cropImageSrc}
          onClose={() => {
            setShowCropper(false);
            setCropImageSrc(null);
          }}
          onCropComplete={handleCropComplete}
          isUploading={isUploading}
        />
      )}
    </>
  );
}
